cmake_minimum_required(VERSION 3.10)
project(DuplexHunter VERSION 0.1 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()

set(CMAKE_CXX_FLAGS_RELEASE "${CMAKE_CXX_FLAGS_RELEASE} -O3")
set(CMAKE_CXX_FLAGS_DEBUG "${CMAKE_CXX_FLAGS_DEBUG} -O0 -g -fsanitize=address")
if(CMAKE_BUILD_TYPE STREQUAL "Debug")
    add_compile_options(-fsanitize=address)
    add_link_options(-fsanitize=address)
endif()

set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${PROJECT_SOURCE_DIR}/out)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${PROJECT_SOURCE_DIR}/out)
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY ${PROJECT_SOURCE_DIR}/out)

find_library(XXHASH_LIB xxhash HINTS /usr/lib /usr/lib/x86_64-linux-gnu)
if(NOT XXHASH_LIB)
    message(FATAL_ERROR "xxHash library not found.")
endif()

find_package(PkgConfig REQUIRED)
pkg_check_modules(LIBMAGIC REQUIRED IMPORTED_TARGET libmagic)

option(DUPLEXHUNTER_WITH_FFMPEG "Enable media analysis using FFmpeg" ON)
if(DUPLEXHUNTER_WITH_FFMPEG)
    pkg_check_modules(FFMPEG IMPORTED_TARGET libavformat libavcodec libavutil libswscale libswresample)
    if(NOT FFMPEG_FOUND)
        message(WARNING "FFmpeg libraries not found. Media analysis disabled.")
        set(DUPLEXHUNTER_WITH_FFMPEG OFF)
    endif()
endif()

option(DUPLEXHUNTER_WITH_UPLOAD "Enable uploading results to the web UI using libcurl and libzip" ON)
if(DUPLEXHUNTER_WITH_UPLOAD)
    pkg_check_modules(CURL IMPORTED_TARGET libcurl)
    pkg_check_modules(LIBZIP IMPORTED_TARGET libzip)
    if(NOT CURL_FOUND OR NOT LIBZIP_FOUND)
        message(WARNING "libcurl or libzip not found. Uploading results disabled.")
        set(DUPLEXHUNTER_WITH_UPLOAD OFF)
    endif()
endif()

add_subdirectory(duplexhunter-lib)
add_subdirectory(duplexhunter-cli)
